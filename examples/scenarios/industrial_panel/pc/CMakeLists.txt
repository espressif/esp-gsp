cmake_minimum_required(VERSION 3.20)
project(industrial_backend LANGUAGES C)
if(NOT ESP_GSP_COMPONENT_DIR)
    if(DEFINED ENV{ESP_GSP_COMPONENT_DIR})
        set(ESP_GSP_COMPONENT_DIR "$ENV{ESP_GSP_COMPONENT_DIR}")
    else()
        foreach(candidate IN ITEMS "../managed_components/espressif__esp-gsp" "../components/espressif__esp-gsp" "../../../..")
            get_filename_component(candidate "${CMAKE_CURRENT_LIST_DIR}/${candidate}" ABSOLUTE)
            if(EXISTS "${candidate}/tools/sim_bridge/CMakeLists.txt")
                set(ESP_GSP_COMPONENT_DIR "${candidate}")
                break()
            endif()
        endforeach()
    endif()
endif()
set(ESP_GSP_EXAMPLE_ASSETS_SUBTREE "scenarios/industrial_panel")
set(ESP_GSP_EXAMPLE_ASSETS_SENTINEL "scenes/assets/production_cell.png")
set(ESP_GSP_EXAMPLE_ASSETS_DESTINATION "${CMAKE_CURRENT_LIST_DIR}/..")
add_subdirectory("${ESP_GSP_COMPONENT_DIR}/tools/sim_bridge" sim_bridge)
gsp_add_backend(industrial_backend SOURCES ../main/industrial_ui.c platform_pc.c INCLUDE_DIRECTORIES ../main SCENES ../scenes/industrial.json)
