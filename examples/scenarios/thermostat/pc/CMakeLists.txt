cmake_minimum_required(VERSION 3.20)
project(thermostat_backend LANGUAGES C)
if(NOT ESP_GSP_COMPONENT_DIR AND DEFINED ENV{ESP_GSP_COMPONENT_DIR})
    set(ESP_GSP_COMPONENT_DIR "$ENV{ESP_GSP_COMPONENT_DIR}")
endif()
if(NOT ESP_GSP_COMPONENT_DIR)
    foreach(relative IN ITEMS "../managed_components/espressif__esp-gsp" "../components/espressif__esp-gsp" "../../../..")
        get_filename_component(candidate "${CMAKE_CURRENT_LIST_DIR}/${relative}" ABSOLUTE)
        if(EXISTS "${candidate}/tools/sim_bridge/CMakeLists.txt")
            set(ESP_GSP_COMPONENT_DIR "${candidate}")
            break()
        endif()
    endforeach()
endif()
if(NOT EXISTS "${ESP_GSP_COMPONENT_DIR}/tools/sim_bridge/CMakeLists.txt")
    message(FATAL_ERROR "Install espressif/esp-gsp or set ESP_GSP_COMPONENT_DIR")
endif()
set(ESP_GSP_EXAMPLE_ASSETS_SUBTREE "scenarios/thermostat")
set(ESP_GSP_EXAMPLE_ASSETS_SENTINEL "scenes/assets/dial.svg")
set(ESP_GSP_EXAMPLE_ASSETS_DESTINATION "${CMAKE_CURRENT_LIST_DIR}/..")
add_subdirectory("${ESP_GSP_COMPONENT_DIR}/tools/sim_bridge" sim_bridge)
gsp_add_backend(thermostat_backend SOURCES ../main/thermostat_ui.c platform_pc.c
    INCLUDE_DIRECTORIES ../main SCENES ../scenes/thermostat.json)
